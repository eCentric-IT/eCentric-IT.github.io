Arquitetura

Os dois planos que se espelham: SDK e API

Cada recurso existe em dois planos que se espelham. O SDK é o que o desenvolvedor usa. A API é o que a DLL expõe. A comunicação entre eles é feita exclusivamente pela ABI C.

Definir os Eventos

Notificações assíncronas da API para o SDK — se houver

Eventos são notificações assíncronas da API para o SDK. São identificados por um inteiro único na faixa reservada ao recurso.

⚡
Os dois arquivos devem estar sempre sincronizados. O comentário no arquivo de API indica isso explicitamente.

Em ecentrix.core.api.pas

Pascal — API 
{ Foo range (400..499): foo lifecycle events
} EVT_FOO_READY     = 400;  { Foo is ready — no payload }
  EVT_FOO_DATA      = 401;  { Data received — payload is UTF-8 text }
  EVT_FOO_TERMINATE = 402;  { Foo terminated — payload is exit code decimal }

Em ecentric.sdk.pas

Pascal — SDK 
{ Process range (200..299) ...
  Foo range (400..499): foo lifecycle events
} EVT_FOO_READY     = 400;
  EVT_FOO_DATA      = 401;
  EVT_FOO_TERMINATE = 402;

Callbacks, IECFoo e TECFoo

2.1 Callbacks

Pascal — SDK
{ Callback for data received from a Foo object
} TECOnFooData = Procedure(Const ASender: IECFoo; Const AData: String) Of Object;

{ Callback for Foo termination
} TECOnFooTerminate = Procedure(Const ASender: IECFoo; Const ACode: Integer) Of Object;

2.2 Interface pública

Herda de IECObject. Expõe apenas o que o desenvolvedor precisa ver.

04

Implementar a Classe no SDK

DoEvent, métodos de operação e late binding

4.1 DoEvent — roteamento de eventos

  • SetString(LRaw, PAnsiChar(AData), ASize) — sempre usar tamanho explícito, nunca strlen
  • UTF8ToString(LRaw) — payload de texto é sempre UTF-8
  • Try..Except vazio em volta do callback — nunca propagar exceções do consumidor
  • Else Inherited — delegar eventos desconhecidos para a base
Pascal — SDK DoEvent
Procedure TECFoo.DoEvent(Const AEventID: Integer; Const AData: Pointer; Const ASize: Cardinal);
Var
  LData: String;
  LCode: Integer;
Begin
  Case AEventID Of

    EVT_FOO_DATA:
    Begin
      If Assigned(FOnData) And (ASize > 0) Then
      Begin
         Var LRaw: RawByteString;
         SetString(LRaw, PAnsiChar(AData), ASize);
         LData := UTF8ToString(LRaw);
         Try
            FOnData(Self, LData);
         Except
         End;
      End;
    End;

    EVT_FOO_TERMINATE:
    Begin
      If Assigned(FOnTerminate) Then
      Begin
         If ASize > 0 Then
         Begin
            Var LRaw: RawByteString;
            SetString(LRaw, PAnsiChar(AData), ASize);
            LCode := StrToIntDef(String(LRaw), 0);
         End
         Else
            LCode := 0;
         Try
            FOnTerminate(Self, LCode);
         Except
         End;
      End;
    End;

  Else
    Inherited DoEvent(AEventID, AData, ASize);
  End;
End;

4.2 Métodos de operação

Pascal — SDK Method
Function TECFoo.DoSomething(Const AParam: String): Boolean;
Var
  LRet: String;

End;

4.3 Late binding

Aplicar quando eventos podem chegar durante o Invoke de criação — antes de o consumidor atribuir os callbacks.

Pascal — Late Binding
Procedure TECFoo.SetOnData(Const ACallback: TECOnFooData);
Begin
  FOnData := ACallback;
{ Late binding: if data arrived before the callback was assigned, deliver now
} If Assigned(FOnData) And (Length(FPendingData) > 0) Then
  Begin
     For I := 0 To High(FPendingData) Do
     Try
        FOnData(Self, FPendingData[I]);
     Except
     End;
     SetLength(FPendingData, 0);
  End;
End;

5.1 AttachFoo

Function TECentric.AttachFoo(Const AHandle: Cardinal): IECFoo;
Begin
  Result := Nil;

     Result := TECFoo.Create(AHandle);

5.2 Foo — caso simples

Pascal — SDK Factory Simple
Function TECentric.Foo(Const AParam: String): IECFoo;
Var
  LRet: String;
  LHandle: Cardinal;
Begin
  Result := Nil;
  LRet := Invoke(0, _EC_ROOT_FOO, JsonParam('param', AParam));
  LHandle := ExtractHandle(LRet);
  If LHandle > 0 Then Begin
     Result := TECFoo.Create(LHandle);
     Result.SetHandle(LHandle);
     RegisterObject(LHandle, Result);
  End;
End;

5.3 Foo — com GSDKPending (eventos durante criação)

Pascal — SDK Factory Pending
Function TECentric.Foo(Const AParam: String): IECFoo;
Var
  LRet: String;
  LHandle: Cardinal;
  LFoo: IECFoo;
  LThreadId: Cardinal;
Begin
  Result := Nil;
  LFoo := TECFoo.Create(0);
  LThreadId := GetCurrentThreadId;
{ Pre-register in pending map so OnCoreEvent can buffer early events
  that arrive during Invoke before the handle is known to the SDK.
} TMonitor.Enter(GSDKLock);
  Try
     GSDKPending.AddOrSetValue(LThreadId, LFoo);
  Finally
     TMonitor.Exit(GSDKLock);
  End;
  Try
     LRet := Invoke(0, _EC_ROOT_FOO, JsonParam('param', AParam));
  Finally
     TMonitor.Enter(GSDKLock);
     Try
        GSDKPending.Remove(LThreadId);
     Finally
        TMonitor.Exit(GSDKLock);
     End;
  End;
  LHandle := ExtractHandle(LRet);
  If LHandle > 0 Then Begin
     LFoo.SetHandle(LHandle);
     RegisterObject(LHandle, LFoo);
     Result := LFoo;
  End;
End;
06

Criar TAPIFoo na API

Declaração, InjectInterface, dispatchers, Close e Destructor

📌
Regra: apenas métodos Published são visíveis pelo RTTI e chamáveis pela ABI. Métodos Public são infraestrutura interna.

6.1 Declaração da classe

Pascal — API Class
{ Foo resource — wraps IECFoo over the TAPIObject infrastructure
} TAPIFoo = Class(TAPIObject)
  Strict Private
    FFoo: IECFoo;             { Core interface — Nil until InjectInterface }
    FClosedFlag: Integer;     { Atomic idempotency flag for Close }
    Procedure DoData(Const AData: String);
    Procedure DoTerminate(Const ACode: Cardinal);
  Public
    Procedure InjectInterface(Const AFoo: IECFoo);
    Destructor Destroy; Override;
  Published
    Function DoSomething(Const AJs: AnsiString): AnsiString;
    Function Close(Const AJs: AnsiString): AnsiString;
  End;

6.2 InjectInterface

Pascal — API InjectInterface
Procedure TAPIFoo.InjectInterface(Const AFoo: IECFoo);
Begin
  FFoo := AFoo;
  FClosedFlag := 0;
  If FFoo <> Nil Then Begin
     FFoo.OnData :=
        Procedure(Const AData: String)
        Begin
           DoData(AData);
        End;
     FFoo.OnTerminate :=
        Procedure(Const ACode: Cardinal)
        Begin
           DoTerminate(ACode);
        End;
  End;
End;

6.3 Dispatchers de evento

  • Capturar Self.Handle em variável local antes de qualquer operação
  • Checar LHandle = 0 antes de despachar
  • Payload texto: serializar como UTF8String
  • Payload numérico: serializar como AnsiString(IntToStr(...))
  • Payload ausente: passar Nil, 0
  • Sempre Try..Except vazio em volta do dispatch
Pascal — API Dispatchers
Procedure TAPIFoo.DoData(Const AData: String);
Var
  LHandle: Cardinal;
  LPayload: UTF8String;
Begin
  If IsShuttingDown Or (GAPIObjects = Nil) Then Exit;
  LHandle := Self.Handle;
  If LHandle = 0 Then Exit;
  LPayload := UTF8String(AData);
  Try
     DispatchCoreEvent(LHandle, EVT_FOO_DATA, Pointer(LPayload), Length(LPayload));
  Except
  End;
End;

Procedure TAPIFoo.DoTerminate(Const ACode: Cardinal);
Var
  LHandle: Cardinal;
  LCode: AnsiString;
Begin
  If IsShuttingDown Or (GAPIObjects = Nil) Then Exit;
  LHandle := Self.Handle;
  If LHandle = 0 Then Exit;
  LCode := AnsiString(IntToStr(ACode));
  Try
     DispatchCoreEvent(LHandle, EVT_FOO_TERMINATE, Pointer(LCode), Length(LCode));
  Except
  End;
End;

6.4 Métodos Published

Pascal — API Published Method
Function TAPIFoo.DoSomething(Const AJs: AnsiString): AnsiString;
Var
  LParams: IECJSon;
  LParam: String;
Begin
  If FFoo = Nil Then
     Exit(ErrorStr('DoSomething failed', 'Foo interface is missing'));
  LParams := ec.JSon.Factory(AJs);
  If LParams = Nil Then
     Exit(ErrorStr('Invalid JSON payload'));
  LParam := LParams.S['param'];
  Try
     If FFoo.DoSomething(LParam) Then
        Result := JsonValue(True) Else
        Result := ErrorStr('DoSomething failed');
  Except
     On E: Exception Do
        Result := ErrorStr('DoSomething exception', E.Message);
  End;
End;

6.5 Close — padrão obrigatório (5 passos)

🔥
POINT OF NO RETURN: DelObject pode destruir Self. Nada após essa linha pode tocar Self, FHandle, FFoo ou qualquer campo de instância.
Atomic idempotency — only the first thread to swap 0 → 1 proceeds.
If AtomicCmpExchange(FClosedFlag, 1, 0) <> 0 Then Exit(StatsStr('Closed'));
Capture handle immediately before any destructive action.
LHandle := Self.Handle;
Release internal interface — Self still alive here.
If FFoo <> Nil Then Try FFoo.Close; Except End;
Build response before map removal.
Result := StatsStr('Closed');
POINT OF NO RETURN — DelObject may destroy Self. No field access beyond this line.
DelObject(LHandle);
Pascal — API Close
Function TAPIFoo.Close(Const AJs: AnsiString): AnsiString;
Var
  LHandle: Cardinal;
Begin
{ 1. Atomic idempotency: only the first thread to swap 0 -> 1 proceeds
} If AtomicCmpExchange(FClosedFlag, 1, 0) <> 0 Then
     Exit(StatsStr('Closed'));
{ 2. Immediate handle capture before any destructive action
} LHandle := Self.Handle;
{ 3. Internal interface release (Self still alive here)
} If FFoo <> Nil Then Try
     FFoo.Close;
  Except
  End;
{ 4. Build response before map removal
} Result := StatsStr('Closed');
{ 5. POINT OF NO RETURN: DelObject may destroy Self. Nothing after this line
  may touch Self, FHandle, FFoo or any instance field.
} DelObject(LHandle);
End;

6.6 Destructor

Pascal — API Destructor
Destructor TAPIFoo.Destroy;
Begin
{ External close: idempotent via FClosedFlag — only runs if Close was not
  called explicitly by the ABI.
} If (AtomicCmpExchange(FClosedFlag, 1, 0) = 0) And (FFoo <> Nil) Then Try
     FFoo.Close;
  Except
  End;
{ Interface release always occurs in the destructor (Delphi side).
  Dereference is internally protected and tolerates Iface = Nil.
} Dereference(FFoo);
  Inherited;
End;

Factory na API

TAPIRootManager.Foo — declaração e implementação

Pascal — API RootManager
Function TAPIRootManager.Foo(Const AJs: AnsiString): AnsiString;
Var
  LParams: IECJSon;
  LParam: String;
  LFooInf: IECFoo;
  LObj: TAPIFoo;
  LHandle: Cardinal;
Begin
  LParams := ec.JSon.Factory(AJs);
  If LParams = Nil Then
     Exit(ErrorStr('Invalid JSON payload'));
  LParam := LParams.S['param'];
  LFooInf := ec.Foo.Create(LParam);
  If LFooInf = Nil Then
     Exit(ErrorStr('Foo creation failed'));
  LObj := TAPIFoo.Create;
  LHandle := 0;
  Try
     LObj.InjectInterface(LFooInf);
     LHandle := MapObject(LObj);
     Result := JsonHandle(LHandle, 'foo');
  Except
     On E: Exception Do Begin
        If LHandle > 0 Then
           DelObject(LHandle);
        Result := ErrorStr('Foo creation exception', E.Message);
     End;
  End;
End;
